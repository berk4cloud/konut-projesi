## 🔄 CACHE SERVICE (In-Memory MVP)

```javascript
// src/services/cache.service.js

import NodeCache from 'node-cache';

const cache = new NodeCache({ 
  stdTTL: 3600,  // 1 hour default
  checkperiod: 120  // cleanup every 2 minutes
});

export const cacheService = {
  async get(key) {
    return cache.get(key) || null;
  },

  async set(key, data, ttl = 3600) {
    cache.set(key, data, ttl);
  },

  async delete(key) {
    cache.del(key);
  },

  async clearPattern(pattern) {
    const keys = cache.keys();
    keys.forEach(key => {
      if (key.includes(pattern)) {
        cache.del(key);
      }
    });
  },

  async connect() {
    console.log('✅ In-memory cache initialized');
  },

  async disconnect() {
    cache.flushAll();
    console.log('✅ Cache flushed');
  }
};
```

## 🐳 Docker Production Setup (İşler Bitince)

### docker-compose.yml (Production)

```yaml
version: '3.9'

services:
  postgres:
    image: postgres:14-alpine
    environment:
      POSTGRES_USER: plan4flex
      POSTGRES_PASSWORD: ${DB_PASSWORD}
      POSTGRES_DB: plan4flex_prod
    volumes:
      - postgres_data:/var/lib/postgresql/data
    networks:
      - plan4flex

  redis:
    image: redis:7-alpine
    command: redis-server --requirepass ${REDIS_PASSWORD}
    volumes:
      - redis_data:/data
    networks:
      - plan4flex

  backend:
    build: ./backend
    environment:
      NODE_ENV: production
      DATABASE_URL: postgresql://plan4flex:${DB_PASSWORD}@postgres:5432/plan4flex_prod
      REDIS_URL: redis://:${REDIS_PASSWORD}@redis:6379
      JWT_SECRET: ${JWT_SECRET}
    ports:
      - "3000:3000"
    depends_on:
      - postgres
      - redis
    networks:
      - plan4flex

  frontend:
    build: ./frontend
    environment:
      VITE_API_URL: http://backend:3000/api
    ports:
      - "80:80"
    networks:
      - plan4flex

volumes:
  postgres_data:
  redis_data:

networks:
  plan4flex:
```
