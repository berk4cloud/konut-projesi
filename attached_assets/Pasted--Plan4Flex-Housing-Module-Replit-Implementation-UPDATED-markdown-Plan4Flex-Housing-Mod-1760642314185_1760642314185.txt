  # PostgreSQL Database
  postgres:
    image: postgres:14-alpine

      POSTGRES_USER: ${DB_USER}
      POSTGRES_PASSWORD: ${DB_PASSWORD}
      POSTGRES_DB: ${DB_NAME}

      - postgres_data:/var/lib/postgresql/data

      retries: 5

      - plan4flex-network

  # Redis Cache

    image: redis:7-alpine

      - redis_data:/data

      - "6379:6379"

      test: ["CMD", "redis-cli", "ping"]

## 🚀 REDIS CACHING STRATEGY

### Cache Service (Backend)

```javascript
// src/services/cache.service.js

import { createClient } from 'redis';

const redisClient = createClient({
  url: process.env.REDIS_URL || 'redis://localhost:6379',
});

export const cacheService = {
  // Get from cache
  async get(key) {
    try {
      const data = await redisClient.get(key);
      return data ? JSON.parse(data) : null;
    } catch (error) {
      console.error('Cache GET error:', error);
      return null;
    }
  },

  // Set cache with TTL
  async set(key, data, ttl = 3600) {
    try {
      await redisClient.setEx(key, ttl, JSON.stringify(data));
    } catch (error) {
      console.error('Cache SET error:', error);
    }
  },

  // Delete from cache
  async delete(key) {
    try {
      await redisClient.del(key);
    } catch (error) {
      console.error('Cache DELETE error:', error);
    }
  },

  // Clear pattern
  async clearPattern(pattern) {
    try {
      const keys = await redisClient.keys(pattern);
      if (keys.length > 0) {
        await redisClient.del(keys);
      }
    } catch (error) {
      console.error('Cache CLEAR error:', error);
    }
  },

  // Connect
  async connect() {
    await redisClient.connect();
  },

  // Disconnect
  async disconnect() {
    await redisClient.disconnect();
  },
};

// Redis Key Patterns
export const CACHE_KEYS = {
  HOUSES: (tenantId) => `houses:${tenantId}`,
  HOUSE_DETAIL: (houseId) => `house:${houseId}`,
  CAPACITY: (tenantId, date) => `capacity:${tenantId}:${date}`,
  WORKERS: (tenantId) => `workers:${tenantId}`,
  RESERVATIONS: (workerId) => `reservations:${workerId}`,
  USER_PROFILE: (userId) => `user:${userId}`,
};
```

### Cache Invalidation

```javascript
// Her update sonrası cache'i temizle

async createReservation(data) {
  const reservation = await prisma.reservation.create({ data });
  
  // Invalidate caches
  await cacheService.delete(CACHE_KEYS.CAPACITY(data.tenantId, data.startDate));
  await cacheService.delete(CACHE_KEYS.HOUSES(data.tenantId));
  await cacheService.clearPattern(`reservations:${data.workerId}:*`);
  
  return reservation;
}
```

### API Controller (with cache)

```javascript
// src/controllers/housing.controller.js

export async function getHouses(req, res) {
  const { tenantId } = req.user;
  const { date } = req.query;

  // Check cache first
  const cacheKey = CACHE_KEYS.HOUSES(tenantId);
  let houses = await cacheService.get(cacheKey);

  if (houses) {
    return res.json({ success: true, data: houses, source: 'cache' });
  }

  // Fetch from database
  houses = await prisma.house.findMany({
    where: { tenantId },
    include: {
      rooms: {
        include: {
          beds: {
            include: { worker: true },
          },
        },
      },
    },
  });

  // Cache for 1 hour
  await cacheService.set(cacheKey, houses, 3600);

  res.json({ success: true, data: houses, source: 'database' });
}
```
